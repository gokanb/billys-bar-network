! ---------- ACLs ----------
ip access-list extended POS-IN
 remark Allow DHCP
 permit udp any any eq 67
 remark Allow DNS and NTP to servers
 permit udp 10.10.1.32 0.0.0.31 host 10.10.1.130 eq 53
 permit udp 10.10.1.32 0.0.0.31 host 10.10.1.132 eq 123
 remark Block all internal networks
 deny ip any 10.10.0.0 0.0.1.255
 remark Allow internet for payment processing
 permit ip 10.10.1.32 0.0.0.31 any
exit
ip access-list extended CCTV-IN
 remark Allow DHCP
 permit udp any any eq 67
 remark NVR can talk to staff (manager viewing footage)
 permit ip host 10.10.1.98 10.10.1.64 0.0.0.31
 remark Block everything else - no internet for cameras
 deny ip any any
exit
ip access-list extended STAFF-IN
 remark MANAGER is the admin PC - can manage network devices
 permit ip host 10.10.1.70 10.10.1.112 0.0.0.15
 deny ip any 10.10.1.112 0.0.0.15
 remark Block staff from POS
 deny ip any 10.10.1.32 0.0.0.31
 remark Allow NVR footage, block cameras
 permit ip any host 10.10.1.98
 deny ip any 10.10.1.96 0.0.0.15
 permit ip any any
exit
ip access-list extended STAFFWIFI-IN
 permit udp any any eq 67
 deny ip any 10.10.1.112 0.0.0.15
 deny ip any 10.10.1.32 0.0.0.31
 deny ip any 10.10.1.96 0.0.0.15
 permit ip any any
exit
ip access-list extended GUEST-IN
 permit udp any any eq 67
 permit udp 10.10.0.0 0.0.0.255 host 10.10.1.130 eq 53
 deny ip any 10.10.0.0 0.0.1.255
 permit ip 10.10.0.0 0.0.0.255 any
exit
